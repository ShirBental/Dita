    // ---------------------------------------------------------------- F3: pure state machine, exhaustive
    @Test void stateMachineInvariantsHoldForEveryInput() {
        for (Validity cur : Validity.values()) for (boolean live : new boolean[]{false, true})
            for (boolean storable : new boolean[]{false, true}) for (boolean wd : new boolean[]{false, true})
                for (boolean rej : new boolean[]{false, true}) {
                    Validity t = ValidityReconciler.target(cur, live, storable, wd, rej);
                    String c = cur + " live=" + live + " v6=" + storable + " wd=" + wd + " rej=" + rej + " -> " + t;
                    if (cur == Validity.UNRESOLVED) assertEquals(Validity.UNRESOLVED, t, "sticky " + c);
                    if (live) assertNotEquals(Validity.ACTIVE, t, "live relation never ACTIVE " + c);
                    if (!storable && cur != Validity.UNRESOLVED) assertNotEquals(Validity.UNRESOLVED, t, "V5 cannot store UNRESOLVED " + c);
                    if (cur != Validity.ACTIVE && t == Validity.ACTIVE)
                        assertTrue(rej && !wd && storable && !live, "revival only by pure rejection on V6 " + c);
                    if (wd && cur == Validity.SUPERSEDED && storable) assertEquals(Validity.UNRESOLVED, t, c);
                }
    }

    // ---------------------------------------------------------------- F3: chained / fan-in / reconfirmation
    static InMemoryMessageStore chain() {
        var s = new InMemoryMessageStore();
        s.append(G, m("a", 0, null, "Dinner is at 7 PM."));
        s.append(G, m("b", 60, null, "Correction: dinner is at 8 PM."));
        s.append(G, m("c", 120, null, "Update: dinner is at 9 PM."));
        extract(s, List.of(p("a", "dinner", "Dinner", "Dinner is at 7 PM."), p("b", "dinner", "dinner", "dinner is at 8 PM."),
                p("c", "dinner", "dinner", "dinner is at 9 PM.")), "a", "b", "c");
        return s;
    }

    @Test void chainedCorrection_withdrawTheLatest_noOlderClaimRevives() {
        var s = chain();
        System.out.println("chain relations: " + supersedesKeys(s));
        assertEquals(Validity.ACTIVE, by(s, "c", "dinner").assertion().validity());
        changes(s).delete(G, "c", T.plusSeconds(200), DeletionSource.FORGET_COMMAND);
        System.out.println("after delete c: a=" + by(s, "a", "dinner").assertion().validity() + " b=" + by(s, "b", "dinner").assertion().validity());
        assertNotEquals(Validity.ACTIVE, by(s, "a", "dinner").assertion().validity());
        assertNotEquals(Validity.ACTIVE, by(s, "b", "dinner").assertion().validity());
        for (boolean limited : new boolean[]{false, true}) {
            GroundedAnswer ans = ask(s, limited, "What time is dinner?");
            assertFalse(ans.text().matches("(?s).*\\b[789] PM.*"), "limited=" + limited + " → " + ans.text());
        }
    }

    @Test void chainedCorrection_withdrawTheMiddle_thenRejectEveryRemainingEdge_neverRevivesAWithdrawnTarget() {
        var s = chain();
        changes(s).delete(G, "b", T.plusSeconds(200), DeletionSource.FORGET_COMMAND);
        Validity a1 = by(s, "a", "dinner").assertion().validity();
        System.out.println("after delete b: a=" + a1 + " c=" + by(s, "c", "dinner").assertion().validity() + " rel=" + supersedesKeys(s));
        reject(s, supersedesKeys(s));
        // a was the target of a withdrawn correction (b) → must not become ACTIVE through rejection
        assertEquals(Validity.UNRESOLVED, by(s, "a", "dinner").assertion().validity());
        assertEquals(Validity.ACTIVE, by(s, "c", "dinner").assertion().validity());
        GroundedAnswer ans = ask(s, false, "What time is dinner?");
        assertFalse(ans.text().contains("7 PM") || ans.text().contains("8 PM"), ans.text());
    }

    @Test void fanIn_rejectFalseEdgeFirst_thenWithdrawGenuine_staysUnresolved() {
        var s = new InMemoryMessageStore();
        s.append(G, m("a", 0, null, "Dinner is at 7 PM."));
        s.append(G, m("b", 60, null, "Correction: dinner is at 8 PM."));
        s.append(G, m("c", 90, null, "The music starts at nine."));
        extract(s, List.of(p("a", "dinner", "Dinner", "Dinner is at 7 PM."), p("b", "dinner", "dinner", "dinner is at 8 PM."),
                p("c", "music", "The music", "The music starts at nine.")), "a", "b", "c");
        String a = by(s, "a", "dinner").id(), c = by(s, "c", "music").id();
        s.addRelation(G, new KnowledgeRelation(G.groupId(), c, a, KnowledgeRelation.Kind.SUPERSEDES, "c"));
        ValidityReconciler.reconcile(s, G);
        var falseEdge = s.relations(G).stream().filter(r -> r.fromId().equals(c)).map(KnowledgeRelation::key).toList();
        reject(s, falseEdge);
        assertEquals(Validity.SUPERSEDED, by(s, "a", "dinner").assertion().validity(), "genuine edge still live");
        changes(s).delete(G, "b", T.plusSeconds(120), DeletionSource.FORGET_COMMAND);
        assertEquals(Validity.UNRESOLVED, by(s, "a", "dinner").assertion().validity());
        assertFalse(ask(s, false, "What time is dinner?").text().contains("7 PM"));
    }

    @Test void editedCorrection_thenReconfirmationWithIdenticalWordingIsAnswerable_andRepairableLater() {
        var s = dinner(false);
        changes(s).edit(G, "b", "Never mind, dinner time is open.", T.plusSeconds(120));
        assertEquals(Validity.UNRESOLVED, by(s, "a", "dinner").assertion().validity());
        assertFalse(ask(s, false, "What time is dinner?").text().contains("7 PM"));
        s.append(G, m("d", 400, null, "Dinner is at 7 PM."));
        extract(s, List.of(p("d", "dinner", "Dinner", "Dinner is at 7 PM.")), "d");
        String d = by(s, "d", "dinner").id();
        assertEquals(Validity.ACTIVE, by(s, "d", "dinner").assertion().validity());
        assertEquals(Validity.UNRESOLVED, by(s, "a", "dinner").assertion().validity(), "old claim not revived by reconfirmation");
        GroundedAnswer ans = ask(s, false, "What time is dinner?");
        assertTrue(ans.text().contains("7 PM") && cited(ans).contains("d") && !cited(ans).contains("a"), ans.text() + " " + cited(ans));
        // a fresh V6 false relation on the reconfirmation is repairable
        s.append(G, m("e", 500, null, "The band arrives at six."));
        extract(s, List.of(p("e", "band", "The band", "The band arrives at six.")), "e");
        s.addRelation(G, new KnowledgeRelation(G.groupId(), by(s, "e", "band").id(), d, KnowledgeRelation.Kind.SUPERSEDES, "e"));
        ValidityReconciler.reconcile(s, G);
        assertEquals(Validity.SUPERSEDED, by(s, "d", "dinner").assertion().validity());
        reject(s, s.relations(G).stream().filter(r -> r.toId().equals(d)).map(KnowledgeRelation::key).toList());
        assertEquals(Validity.ACTIVE, by(s, "d", "dinner").assertion().validity());
        assertEquals(Validity.UNRESOLVED, by(s, "a", "dinner").assertion().validity());
    }

    // ---------------------------------------------------------------- F2: strict serialization in-memory
    interface Mutation { void run(MessageStore truth); }

    private void strictGate(Mutation mutation, boolean expectStaleImpossible, String label) throws Exception {
        var truth = dinner(false);
        truth.append(G, m("x", 90, null, "Parking is free."));
        var paused = new CountDownLatch(1); var release = new CountDownLatch(1); var hit = new AtomicBoolean();
        MessageStore view = (MessageStore) Proxy.newProxyInstance(MessageStore.class.getClassLoader(), new Class<?>[]{MessageStore.class},
                (proxy, method, args) -> {
                    try {
                        boolean gate = Arrays.stream(Thread.currentThread().getStackTrace()).anyMatch(f -> f.getMethodName().equals("currentAtReturn"));
                        // pause BETWEEN verifier and evidence-view: on the first knowledge() read inside the gate
                        if (gate && method.getName().equals("knowledge") && hit.compareAndSet(false, true)) {
                            paused.countDown();
                            if (!release.await(5, TimeUnit.SECONDS)) throw new AssertionError("not released");
                        }
                        return method.invoke(truth, args);
                    } catch (InvocationTargetException e) { throw e.getCause(); }
                });
        var pool = Executors.newFixedThreadPool(2);
        try {
            var answering = pool.submit(() -> query(view, (g, q, k) -> List.of(new Hit("b", 1, "direct", false)), true)
                    .ask(G, "viewer", "What time is dinner?"));
            assertTrue(paused.await(5, TimeUnit.SECONDS), label + ": reached gate");
            var writing = pool.submit(() -> mutation.run(truth));
            boolean blocked;
            try { writing.get(400, TimeUnit.MILLISECONDS); blocked = false; } catch (TimeoutException e) { blocked = true; }
            assertTrue(blocked, label + ": writer must not commit while the gate holds the group");
            release.countDown();
            GroundedAnswer a = answering.get(5, TimeUnit.SECONDS);
            writing.get(5, TimeUnit.SECONDS);
            assertTrue(a.found() && a.text().contains("8 PM"), label + ": answer valid at its serialization point → " + a.text());
            GroundedAnswer next = ask(truth, true, "What time is dinner?");
            if (expectStaleImpossible) assertFalse(next.text().contains("8 PM") || next.text().contains("7 PM"), label + " next → " + next.text());
        } finally { release.countDown(); pool.shutdownNow(); }
    }

    @Test void gateHoldsAgainstEditOfCitedCorrection() throws Exception {

    }
    @Test void gateHoldsAgainstDeletionOfCitedCorrection() throws Exception {
        strictGate(t -> changes(t).delete(G, "b", T.plusSeconds(180), DeletionSource.FORGET_COMMAND), true, "delete");
    }
    @Test void gateHoldsAgainstNewRelationRetiringTheCitedClaim() throws Exception {

    }

    /** Pre-gate withdrawal (during retrieval): the retry must not fall back to quoting the withdrawn/unresolved 7 PM. */
    @Test void preGateWithdrawal_retryNeverRevivesTheOlderClaim() {
        for (boolean limited : new boolean[]{false, true}) {
            var truth = dinner(false);
            var once = new AtomicBoolean();
            var attempts = new AtomicInteger();
            // withdraw b during the first READ of b's knowledge or retrieval, whichever comes first
            MessageStore view = (MessageStore) Proxy.newProxyInstance(MessageStore.class.getClassLoader(), new Class<?>[]{MessageStore.class},
                    (proxy, method, args) -> {
                        try {
                            Object r = method.invoke(truth, args);
                            boolean gate = Arrays.stream(Thread.currentThread().getStackTrace()).anyMatch(f -> f.getMethodName().equals("currentAtReturn"));
                            if (!gate && method.getName().equals("find") && "b".equals(args[1]) && once.compareAndSet(false, true))
                                changes(truth).delete(G, "b", T.plusSeconds(150), DeletionSource.FORGET_COMMAND);
                            return r;
                        } catch (InvocationTargetException e) { throw e.getCause(); }
                    });
            GroundedAnswer a = query(view, (g, q, k) -> {
                attempts.incrementAndGet();
                return List.of(new Hit("b", 1, "direct", false), new Hit("a", 0.9, "direct", false));
            }, limited).ask(G, "viewer", "What time is dinner?");
            assertTrue(once.get(), "withdrawal happened");
            assertTrue(truth.find(G, "b").orElseThrow().deleted());
            System.out.println("pre-gate limited=" + limited + " attempts=" + attempts.get() + " → " + a.text().replace('\n', ' '));
            assertFalse(a.text().contains("7 PM") || a.text().contains("8 PM"), "limited=" + limited + " → " + a.text());
            assertTrue(attempts.get() <= 2, "bounded: " + attempts.get());
        }
    }

    /** In-memory note: the monitor is store-wide, so another group's writer is ALSO serialized (no data mixing). */
    @Test void inMemory_otherGroupWriterIsSerializedButUnaffected() throws Exception {
        var truth = dinner(false);
        GroupScope h = new GroupScope("other047");
        truth.append(h, new Message("z", h.groupId(), "bo", T, null, "Other group text.", SourceKind.SYNTHETIC, false));
        var paused = new CountDownLatch(1); var release = new CountDownLatch(1); var hit = new AtomicBoolean();
        MessageStore view = (MessageStore) Proxy.newProxyInstance(MessageStore.class.getClassLoader(), new Class<?>[]{MessageStore.class},
                (proxy, method, args) -> {
                    try {
                        boolean gate = Arrays.stream(Thread.currentThread().getStackTrace()).anyMatch(f -> f.getMethodName().equals("currentAtReturn"));
                        if (gate && method.getName().equals("knowledge") && hit.compareAndSet(false, true)) { paused.countDown(); release.await(5, TimeUnit.SECONDS); }
                        return method.invoke(truth, args);
                    } catch (InvocationTargetException e) { throw e.getCause(); }
                });
        var pool = Executors.newFixedThreadPool(2);
        try {
            var answering = pool.submit(() -> query(view, (g, q, k) -> List.of(new Hit("b", 1, "direct", false)), true).ask(G, "viewer", "What time is dinner?"));
            assertTrue(paused.await(5, TimeUnit.SECONDS));
            var writing = pool.submit(() -> changes(truth).delete(h, "z", T.plusSeconds(10), DeletionSource.FORGET_COMMAND));
            boolean blocked;
            try { writing.get(300, TimeUnit.MILLISECONDS); blocked = false; } catch (TimeoutException e) { blocked = true; }
            System.out.println("in-memory other-group writer blocked during gate: " + blocked);
            release.countDown();
            assertTrue(answering.get(5, TimeUnit.SECONDS).text().contains("8 PM"));
            writing.get(5, TimeUnit.SECONDS);
            assertTrue(truth.find(h, "z").orElseThrow().deleted());
        } finally { release.countDown(); pool.shutdownNow(); }
    }
